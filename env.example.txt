# Skopiuj zawartość do pliku .env i uzupełnij własnymi danymi.
# NIE commituj pliku .env do repozytorium!
#
# .env zawiera OPCJE WSPÓLNE. Firmy (tokeny + NIP-y) konfiguruj w firmy.json
# (wzór: firmy.example.json) — wtedy skrypt zapyta, dla której firmy pobrać.

# --- Pojedyncza firma (używane TYLKO gdy nie ma firmy.json) ---
# Token KSeF z Aplikacji Podatnika (kontekst Twojej firmy)
KSEF_TOKEN=token
# NIP firmy (kontekst uwierzytelnienia)
KSEF_NIP=1234567890
# Środowisko dla firmy z .env: prod | demo | test
KSEF_ENV=prod

# --- Opcje wspólne ---
# Typ daty filtrowania: Issue (wystawienia) | Invoicing (przyjęcia w KSeF)
# | PermanentStorage (trwałego zapisu). Domyślnie Issue.
KSEF_DATE_TYPE=Issue

# Tryb pobierania:
#   single  – PRZYROSTOWO: pobiera listę faktur, sprawdza co masz na dysku
#             i ściąga TYLKO nowe (zalecane do regularnego użytku), domyślny
#   export  – MASOWO: pobiera całą zaszyfrowaną paczkę z całego zakresu
#             (do jednorazowego ściągnięcia dużej historii)
KSEF_MODE=single

# Format zapisu faktur:
#   xml  – oryginalny plik XML (domyślnie)
#   pdf  – wizualizacja PDF oficjalnym generatorem MF (do druku; wymaga Node.js)
KSEF_FORMAT=pdf

# --- Uruchomienie nieinteraktywne (cron) ---
# Zakres dat (RRRR-MM-DD) — używany, gdy skrypt NIE pyta interaktywnie.
# Interaktywnie skrypt pyta o daty (Enter = nowe z bieżącego miesiąca).
