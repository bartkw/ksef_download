# Token KSeF wygenerowany w Aplikacji Podatnika KSeF (kontekst Twojej firmy)
KSEF_TOKEN=token

# NIP firmy (kontekst uwierzytelnienia)
KSEF_NIP=8792408754

# Środowisko: prod | demo | test
KSEF_ENV=prod

# Zakres dat (format YYYY-MM-DD). Jeśli pominiesz, weźmie ostatnie 30 dni.
# Zakres może być dowolnie szeroki — skrypt dzieli go na okna po <3 mies.
KSEF_DATE_FROM=2026-08-01
KSEF_DATE_TO=2026-08-26

# Typ daty filtrowania: Issue (wystawienia) | Invoicing (przyjęcia w KSeF)
# | PermanentStorage (trwałego zapisu). Domyślnie Issue.
KSEF_DATE_TYPE=Issue

# Tryb pobierania:
#   single  – faktura po fakturze (metadane + GET), domyślny
#   export  – asynchroniczny eksport zaszyfrowanej paczki ZIP (masowo)
KSEF_MODE=export

# Format zapisu faktur:
#   xml  – oryginalny plik XML (domyślnie)
#   pdf  – wizualizacja PDF wg oficjalnego arkusza MF (do druku)
#          Wymaga: pip install lxml weasyprint
KSEF_FORMAT=pdf
